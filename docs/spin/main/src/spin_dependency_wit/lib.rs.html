12/// The exports and middleware status of a Wasm component, decoded once.
13///
14/// Decoding a component's WIT is relatively expensive, so callers that need both
15/// its exports and whether it is middleware should build a single
16/// `ComponentInterfaces` and reuse it rather than re-inspecting the raw bytes.
17pub struct ComponentInterfaces {
18    exports: Vec<String>,
19    is_http_middleware: bool,
20}
21
22impl ComponentInterfaces {
23    /// Decode a Wasm component and inspect its world's imports and exports.
24    pub fn from_component(wasm_bytes: &[u8]) -> anyhow::Result<Self> {
25        let decoded = read_wasm(wasm_bytes)?;
26        let (resolve, world_id) = match &decoded {
27            DecodedWasm::WitPackage(resolve, pkg_id) => {
28                let world_id = resolve.select_world(&[*pkg_id], None)?;
29                (resolve, world_id)
30            }
31            DecodedWasm::Component(resolve, world_id) => (resolve, *world_id),
32        };
33
34        let world = resolve
35            .worlds
36            .get(world_id)
37            .context("world not found in decoded Wasm")?;
38
39        let exports = world
40            .exports
41            .keys()
42            .filter_map(|key| format_world_key(resolve, key))
43            .collect();
44
45        // Middleware both imports and exports `wasi:http/handler`: the export
46        // receives the request; the import forwards it to the next component in
47        // the pipeline. A regular HTTP handler only exports `handler`.
48        let is_http_middleware = contains_wasi_http_handler(resolve, world.imports.keys())
49            && contains_wasi_http_handler(resolve, world.exports.keys());
50
51        Ok(Self {
52            exports,
53            is_http_middleware,
54        })
55    }
56
57    /// Whether the component is HTTP middleware.
58    pub fn is_http_middleware(&self) -> bool {
59        self.is_http_middleware
60    }
61
62    /// The component's exports, formatted as dependency-name strings.
63    ///
64    /// Interface exports from a package are formatted as
65    /// `namespace:package/interface@version`. Exports declared with a bare name
66    /// (for example an exported function, or an inline interface) are returned as
67    /// that plain kebab-case name.
68    pub fn exports(&self) -> &[String] {
69        &self.exports
70    }
71}

73/// Format a world key as a dependency-name string, returning `None` for
74/// interface exports that lack package information.
75fn format_world_key(resolve: &wit_parser::Resolve, key: &wit_parser::WorldKey) -> Option<String> {
76    match key {
77        wit_parser::WorldKey::Name(name) => Some(name.clone()),
78        wit_parser::WorldKey::Interface(iface_id) => {
79            let iface = resolve.interfaces.get(*iface_id)?;
80            let pkg = resolve.packages.get(iface.package?)?;
81            let ns = &pkg.name.namespace;
82            let name = &pkg.name.name;
83            let iface_name = iface.name.as_deref().unwrap_or("unknown");
84            Some(match &pkg.name.version {
85                Some(version) => format!("{ns}:{name}/{iface_name}@{version}"),
86                None => format!("{ns}:{name}/{iface_name}"),
87            })
88        }
89    }
90}

92fn contains_wasi_http_handler<'a>(
93    resolve: &wit_parser::Resolve,
94    keys: impl Iterator<Item = &'a wit_parser::WorldKey>,
95) -> bool {
96    keys.filter_map(|key| match key {
97        wit_parser::WorldKey::Interface(id) => resolve.interfaces.get(*id),
98        wit_parser::WorldKey::Name(_) => None,
99    })
100    .any(|iface| {
101        let Some(pkg_id) = iface.package else {
102            return false;
103        };
104        let Some(pkg) = resolve.packages.get(pkg_id) else {
105            return false;
106        };
107        pkg.name.namespace == "wasi"
108            && pkg.name.name == "http"
109            && iface.name.as_deref() == Some("handler")
110    })
111}

113pub async fn extract_wits_into(
114    source: impl ExactSizeIterator<Item = (&DependencyName, &ComponentDependency)>,
115    app_root: impl AsRef<Path>,
116    dest_file: impl AsRef<Path>,
117) -> anyhow::Result<()> {
118    if source.len() == 0 {
119        if dest_file.as_ref().is_file() {
120            _ = tokio::fs::remove_file(dest_file).await; // Not an error if tidying fails
121        }
122        return Ok(());
123    }
124
125    let wit_text = format!(
126        "{}{}",
127        GENERATED_COMMENT,
128        extract_wits(source, app_root).await?
129    );
130
131    tokio::fs::create_dir_all(
132        dest_file
133            .as_ref()
134            .parent()
135            .context("file is root directory")?,
136    )
137    .await?;
138    tokio::fs::write(dest_file, wit_text.as_bytes()).await?;
139
140    Ok(())
141}

149pub async fn extract_wits(
150    source: impl Iterator<Item = (&DependencyName, &ComponentDependency)>,
151    app_root: impl AsRef<Path>,
152) -> anyhow::Result<String> {
153    let loader = WasmLoader::new(app_root.as_ref().to_owned(), None, None).await?;
154
155    let mut package_wits = indexmap::IndexMap::new();
156
157    let mut aggregating_resolve = wit_parser::Resolve::default();
158    let aggregating_pkg_id =
159        aggregating_resolve.push_str("dummy.wit", "package root:component;\n\nworld root {}")?;
160    let aggregating_world_id =
161        aggregating_resolve.select_world(&[aggregating_pkg_id], Some("root"))?;
162
163    // TODO: figure out what to do if we import two itfs from same dep
164    for (index, (dependency_name, dependency)) in source.enumerate() {
165        let import_kind = match dependency_name {
166            DependencyName::Plain(name) => ImportKind::Function(name.clone()),
167            DependencyName::Package(dependency_package_name) => {
168                match &dependency_package_name.interface {
169                    Some(itf) => ImportKind::Interface(itf.clone()),
170                    None => ImportKind::WholePackage,
171                }
172            }
173        };
174
175        let (wasm_path, export) = loader
176            .load_dependency_content(dependency_name, dependency)
177            .await
178            .with_context(|| format!("failed to load dependency {dependency_name}"))?;
179        let wasm_bytes = tokio::fs::read(&wasm_path).await?;
180
181        let decoded = read_wasm(&wasm_bytes)?;
182        let decoded = match export {
183            None => decoded,
184            Some(export) => {
185                munge_aliased_export(decoded, &export, dependency_name).with_context(|| {
186                    format!("failed to map named export {export} to dependency {dependency_name}")
187                })?
188            }
189        };
190        let impo_world = format!("impo-world{index}");
191        let importised = importize(decoded, Some(&impo_world))
192            .with_context(|| format!("failed to map importize dependency {dependency_name}"))?;
193
194        let imports = match &import_kind {
195            ImportKind::WholePackage => all_imports(&importised),
196            ImportKind::Interface(itf) => one_import(&importised, itf.as_ref())?,
197            ImportKind::Function(_) => Default::default(),
198        };
199        let func_import = match &import_kind {
200            ImportKind::Function(f) => one_func_import(&importised, f.as_ref())?,
201            _ => Default::default(),
202        };
203        let type_imports = match &import_kind {
204            ImportKind::Function(_) => world_type_imports(&importised),
205            _ => Default::default(),
206        };
207
208        // Capture WITs for all packages used in the importised thing.
209        // Things like WASI packages may be depended on by multiple packages
210        // so we index on the package name to avoid emitting them twice.
211
212        let root_pkg = importised.package();
213        let useful_pkgs = importised
214            .resolve()
215            .packages
216            .iter()
217            .map(|p| p.0)
218            .filter(|pid| *pid != root_pkg)
219            .collect::<Vec<_>>();
220
221        for p in &useful_pkgs {
222            let pkg_name = importised
223                .resolve()
224                .packages
225                .get(*p)
226                .context("package not found in importised (id lookup failed)")? // shouldn't happen
227                .name
228                .clone();
229            let output = wit_component::OutputToString::default();
230            let mut printer = wit_component::WitPrinter::new(output);
231            printer.print_package(importised.resolve(), *p, false)?;
232            package_wits.insert(pkg_name, printer.output.to_string());
233        }
234
235        // Now add the imports to the aggregating component import world
236
237        let remap = aggregating_resolve.merge(importised.resolve().clone())?;
238        for iid in imports {
239            let mapped_iid = remap.map_interface(iid, Span::default())?;
240            let wk = wit_parser::WorldKey::Interface(mapped_iid);
241            let world_item = wit_parser::WorldItem::Interface {
242                id: mapped_iid,
243                stability: wit_parser::Stability::Unknown,
244                span: Span::default(),
245                docs: Default::default(),
246                external_id: None,
247            };
248            let previous_world_item = aggregating_resolve
249                .worlds
250                .get_mut(aggregating_world_id)
251                .context("aggregated dependency world doesn't exist")? // shouldn't happen
252                .imports
253                .insert(wk, world_item);
254            if let Some(previous_world_item) = previous_world_item {
255                debug_assert!({
256                    let wit_parser::WorldItem::Interface { id, .. } = previous_world_item else {
257                        debug_assert!(
258                            false,
259                            "previous WorldItem did not match inserted WorldItem: not an interface"
260                        );
261                        break;
262                    };
263                    debug_assert_eq!(
264                        id, mapped_iid,
265                        "replacing WorldItem had a different InterfaceId from replaced one"
266                    );
267
268                    true
269                });
270            }
271        }
272        if let Some(mut func) = func_import {
273            // Remap type IDs in the function to reference the aggregating resolve
274            for param in &mut func.params {
275                if let wit_parser::Type::Id(id) = &mut param.ty {
276                    *id = remap.map_type(*id, Span::default())?;
277                }
278            }
279            if let Some(wit_parser::Type::Id(id)) = &mut func.result {
280                *id = remap.map_type(*id, Span::default())?;
281            }
282
283            // Add world-level type definitions that the function depends on
284            let aggregating_world = aggregating_resolve
285                .worlds
286                .get_mut(aggregating_world_id)
287                .context("aggregated dependency world doesn't exist")?;
288            for (name, type_id) in &type_imports {
289                let mapped_id = remap.map_type(*type_id, Span::default())?;
290                let wk = wit_parser::WorldKey::Name(name.clone());
291                let world_item = wit_parser::WorldItem::Type {
292                    id: mapped_id,
293                    span: Span::default(),
294                };
295                aggregating_world.imports.insert(wk, world_item);
296            }
297
298            let wk = wit_parser::WorldKey::Name(func.name.clone());
299            let world_item = wit_parser::WorldItem::Function(func);
300            aggregating_world.imports.insert(wk, world_item);
301        }
302    }
303
304    // Text for the root package and world(s)
305    let world_output = wit_component::OutputToString::default();
306    let mut world_printer = wit_component::WitPrinter::new(world_output);
307    world_printer.print(&aggregating_resolve, aggregating_pkg_id, &[])?;
308
309    let mut buf = String::new();
310
311    // Print the root package and the world(s) with the imports
312    buf.push_str(&world_printer.output.to_string());
313
314    // Print each package
315    for package_wit in package_wits.values() {
316        buf.push_str(package_wit);
317    }
318
319    Ok(buf)
320}

322fn munge_aliased_export(
323    decoded: DecodedWasm,
324    export: &str,
325    new_name: &DependencyName,
326) -> anyhow::Result<DecodedWasm> {
327    let export_qname = spin_serde::DependencyPackageName::try_from(export.to_string())?;
328    let Some(export_itf_name) = export_qname.interface.as_ref() else {
329        anyhow::bail!(
330            "the export name should be a qualified interface name - {export_qname} doesn't specify interface"
331        );
332    };
333    let export_pkg_name = wit_parser::PackageName {
334        namespace: export_qname.package.namespace().to_string(),
335        name: export_qname.package.name().to_string(),
336        version: export_qname.version,
337    };
338
339    let DependencyName::Package(new_name) = new_name else {
340        anyhow::bail!(
341            "the dependency name should be a qualified interface name - {new_name} not qualified"
342        );
343    };
344    let Some(new_itf_name) = new_name.interface.as_ref() else {
345        anyhow::bail!(
346            "the dependency name should be a qualified interface name - {new_name} doesn't specify an interface"
347        );
348    };
349    let new_pkg_name = wit_parser::PackageName {
350        namespace: new_name.package.namespace().to_string(),
351        name: new_name.package.name().to_string(),
352        version: new_name.version.clone(),
353    };
354
355    let (mut resolve, decode_id) = match decoded {
356        DecodedWasm::WitPackage(resolve, id) => (resolve, WorldOrPackageId::Package(id)),
357        DecodedWasm::Component(resolve, id) => (resolve, WorldOrPackageId::World(id)),
358    };
359
360    // Two scenarios:
361    // 1. The new name is in a package that is already in the Resolve
362    //    1a. The package already contains an interface with the right name
363    //    1b. The package does not already contain an interface with the right name
364    // 2. The new name is in a package that is NOT already in the Resolve
365
366    let existing_pkg = resolve
367        .packages
368        .iter()
369        .find(|(_pkg_id, pkg)| pkg.name == new_pkg_name);
370
371    // We address the first level by creating the new-name package if it doesn't exist
372    let (inserting_into_pkg_id, inserting_into_pkg) = match existing_pkg {
373        Some(tuple) => tuple,
374        None => {
375            // insert the needed package
376            let package_wit = format!("package {new_pkg_name};");
377            let pkg_id = resolve
378                .push_str(
379                    std::env::current_dir().context("no current dir")?, // unused
380                    &package_wit,
381                )
382                .with_context(|| format!("failed to create import alias package {new_pkg_name}"))?;
383            let pkg = resolve
384                .packages
385                .get(pkg_id)
386                .context("export alias package created but doesn't exist")?; // shouldn't happen
387            (pkg_id, pkg)
388        }
389    };
390
391    // Second level asks if the new-name package already contains the interface
392    let existing_itf = inserting_into_pkg.interfaces.get(new_itf_name.as_ref());
393    if existing_itf.is_some() {
394        // This makes the questionable assumption that the matchingly interface already
395        // in the package is the same as the export. E.g. given "a:b/i" = { export = "c:d/i" }
396        // where the dep contains an `a:b` package with an interface named `i`, we will generate
397        // from that rather than emitting the `c:d/i` WIT for `a:b/i`. We could be smarter about
398        // this but we don't know if other things depend on `a:b/i` so replacing it could result
399        // in a bad WIT. If this becomes a problem we could maybe try emitting it as something
400        // like `a:b/i-from-c-d` but I'd prefer to cross that bridge when we come to it.
401        return Ok(decode_id.make_decoded_wasm(resolve));
402    }
403
404    // The new-name package does not contain the interface: we need to slurp the EXPORTED itf into the `inserting_into`
405    // package under the NEW (importing) interface name
406    let Some(export_pkg_id) = resolve.package_names.get(&export_pkg_name) else {
407        anyhow::bail!(
408            "export is from a package ({}) that doesn't exist (name lookup failed)",
409            export_pkg_name
410        );
411    };
412    let Some(export_pkg) = resolve.packages.get(*export_pkg_id) else {
413        anyhow::bail!(
414            "export is from a package ({}) that doesn't exist (id lookup failed)",
415            export_pkg_name
416        );
417    };
418    let Some(export_itf_id) = export_pkg.interfaces.get(export_itf_name.as_ref()) else {
419        anyhow::bail!(
420            "export package ({}) doesn't contain interface {} (name lookup failed)",
421            export_pkg_name,
422            export_itf_name
423        );
424    };
425    let Some(export_itf) = resolve.interfaces.get(*export_itf_id) else {
426        anyhow::bail!(
427            "export package ({}) doesn't contain interface {} (id lookup failed)",
428            export_pkg_name,
429            export_itf_name
430        );
431    };
432
433    // Create the new-name interface by cloning the export interface
434    let mut export_itf = export_itf.clone();
435    export_itf.package = Some(inserting_into_pkg_id);
436    export_itf.name = Some(new_itf_name.to_string());
437
438    // Add the new-name interface to the resolve and to the new-name package
439    let export_itf_id_new = resolve.interfaces.alloc(export_itf);
440    let inserting_into_pkg_mut = resolve
441        .packages
442        .get_mut(inserting_into_pkg_id)
443        .context("package id lookup that succeeded before failed now")?; // we re-lookup to get around a "mutable borrow at the same time as immutable borrow" woe
444    inserting_into_pkg_mut
445        .interfaces
446        .insert(new_itf_name.to_string(), export_itf_id_new);
447
448    let decoded = decode_id.make_decoded_wasm(resolve);
449
450    Ok(decoded)
451}

458impl WorldOrPackageId {
459    pub fn make_decoded_wasm(&self, resolve: wit_parser::Resolve) -> DecodedWasm {
460        match self {
461            Self::Package(id) => DecodedWasm::WitPackage(resolve, *id),
462            Self::World(id) => DecodedWasm::Component(resolve, *id),

464    }

467fn all_imports(wasm: &DecodedWasm) -> Vec<wit_parser::InterfaceId> {
468    wasm.resolve()
469        .worlds
470        .iter()
471        .flat_map(|(_wid, w)| w.imports.values())
472        .flat_map(as_interface)
473        .collect()
474}

476fn as_interface(wi: &wit_parser::WorldItem) -> Option<wit_parser::InterfaceId> {
477    match wi {
478        wit_parser::WorldItem::Interface { id, .. } => Some(*id),
479        _ => None,
480    }
481}

483fn as_func(wi: &wit_parser::WorldItem) -> Option<&wit_parser::Function> {
484    match wi {
485        wit_parser::WorldItem::Function(func) => Some(func),
486        _ => None,
487    }

490fn one_import(wasm: &DecodedWasm, name: &str) -> anyhow::Result<Vec<wit_parser::InterfaceId>> {
491    fn find_interface(
492        resolve: &wit_parser::Resolve,
493        iid: wit_parser::InterfaceId,
494    ) -> Option<(wit_parser::InterfaceId, &wit_parser::Interface)> {
495        resolve.interfaces.get(iid).map(|itf| (iid, itf))
496    }
497
498    let id = wasm
499        .resolve()
500        .worlds
501        .iter()
502        .flat_map(|(_wid, w)| w.imports.values())
503        .flat_map(as_interface)
504        .flat_map(|iid| find_interface(wasm.resolve(), iid))
505        .find(|itf| itf.1.name.as_deref() == Some(name))
506        .map(|t| t.0)
507        .with_context(|| format!("interface {name} not found in component binary"))?;
508    Ok(vec![id])
509}
510
511fn world_type_imports(wasm: &DecodedWasm) -> Vec<(String, wit_parser::TypeId)> {
512    wasm.resolve()
513        .worlds
514        .iter()
515        .flat_map(|(_wid, w)| {
516            w.imports.iter().filter_map(|(wk, wi)| {
517                if let wit_parser::WorldItem::Type { id, .. } = wi {
518                    let name = match wk {
519                        wit_parser::WorldKey::Name(n) => n.clone(),
520                        wit_parser::WorldKey::Interface(_) => return None,
521                    };
522                    Some((name, *id))
523                } else {
524                    None
525                }
526            })
527        })
528        .collect()
529}

531fn one_func_import(wasm: &DecodedWasm, name: &str) -> anyhow::Result<Option<wit_parser::Function>> {
532    let funcs = wasm
533        .resolve()
534        .worlds
535        .iter()
536        .flat_map(|w| {
537            w.1.imports
538                .iter()
539                .flat_map(|(_wk, wi)| as_func(wi))
540                .filter(|f| f.name == name)
541        })
542        .collect::<Vec<_>>();
543
544    // This shouldn't happen because we are using the compiled Wasm so there should
545    // be only one world in play.  But belt and braces.
546    if funcs.len() > 1 {
547        anyhow::bail!("Dependency exports more than one function named {name}");
548    }
549    Ok(funcs.first().cloned().cloned())
550}

552fn read_wasm(wasm_bytes: &[u8]) -> anyhow::Result<DecodedWasm> {
553    if wasmparser::Parser::is_component(wasm_bytes) {
554        wit_component::decode(wasm_bytes)
555    } else {
556        let (wasm, bindgen) = wit_component::metadata::decode(wasm_bytes)?;
557        if wasm.is_none() {
558            anyhow::bail!(
559                "input is a core wasm module with no `component-type*` \
560                    custom sections meaning that there is no WIT information; \
561                    is the information not embedded or is this supposed \
562                    to be a component?"
563            )
564        }
565        Ok(DecodedWasm::Component(bindgen.resolve, bindgen.world))
566    }
567}

569fn importize(decoded: DecodedWasm, out_world_name: Option<&String>) -> anyhow::Result<DecodedWasm> {
570    let (mut resolve, world_id) = match decoded {
571        DecodedWasm::Component(resolve, world) => (resolve, world),
572        DecodedWasm::WitPackage(resolve, id) => {
573            let world = resolve.select_world(&[id], None)?;
574            (resolve, world)
575        }
576    };
577
578    resolve
579        .importize(world_id, out_world_name.cloned())
580        .context("failed to move world exports to imports")?;
581
582    Ok(DecodedWasm::Component(resolve, world_id))
583}
