26/// App manifest
27#[derive(Clone, Debug, Serialize, Deserialize, JsonSchema)]
28#[serde(deny_unknown_fields)]
29pub struct AppManifest {
30    /// `spin_manifest_version = 2`
31    #[schemars(with = "usize", range(min = 2, max = 2))]
32    pub spin_manifest_version: FixedVersion<2>,
33    /// `[application]`
34    pub application: AppDetails,
35    /// Application configuration variables. These can be set via environment variables, or
36    /// from sources such as Hashicorp Vault or Azure KeyVault by using a runtime config file.
37    /// They are not available directly to components: use a component variable to ingest them.
38    ///
39    /// Learn more: https://spinframework.dev/variables, https://spinframework.dev/dynamic-configuration#application-variables-runtime-configuration
40    #[serde(default, skip_serializing_if = "Map::is_empty")]
41    pub variables: Map<LowerSnakeId, Variable>,
42    /// The triggers to which the application responds. Most triggers can appear
43    /// multiple times with different parameters: for example, the `http` trigger may
44    /// appear multiple times with different routes, or the `redis` trigger with
45    /// different channels.
46    ///
47    /// Example: `[[trigger.http]]`
48    #[serde(rename = "trigger")]
49    #[serde(default)]
50    #[schemars(with = "json_schema::TriggerSchema")]
51    pub triggers: Map<String, Vec<Trigger>>,
52    /// `[component.<id>]`
53    #[serde(rename = "component")]
54    #[serde(default, skip_serializing_if = "Map::is_empty")]
55    pub components: Map<KebabId, Component>,
56}
57
58impl AppManifest {
59    /// Ensures that required fields (where not enforced in the schema) are populated.
60    pub fn validate_required_fields(&self) -> anyhow::Result<()> {
61        if self.triggers.is_empty() {
62            anyhow::bail!("The application must have at least one trigger");
63        }
64
65        Ok(())
66    }
67
68    /// This method ensures that the dependencies of each component are valid.
69    pub fn validate_dependencies(&self) -> anyhow::Result<()> {
70        for (component_id, component) in &self.components {
71            component
72                .dependencies
73                .validate()
74                .with_context(|| format!("component {component_id:?} has invalid dependencies"))?;
75        }
76        Ok(())
77    }
78
79    /// Whether any component in the application defines the given profile.
80    /// Not every component defines every profile, and components intentionally
81    /// fall back to the anonymouse profile if they are asked for a profile
82    /// they don't define. So this can be used to detect that a user might have
83    /// mistyped a profile (e.g. `spin up --profile deugb`).
84    pub fn ensure_profile(&self, profile: Option<&str>) -> anyhow::Result<()> {
85        let Some(p) = profile else {
86            return Ok(());
87        };
88
89        let is_defined = self.components.values().any(|c| c.profile.contains_key(p));
90
91        if is_defined {
92            Ok(())
93        } else {
94            Err(anyhow!("Profile {p} is not defined in this application"))
95        }
96    }
97}

334        // Specifying a dependency name as a plain-name without a package is an error
335        assert!(
336            ComponentDependencies::deserialize(toml! {
337                "plain-name" = "0.1.0"
338            })
339            .unwrap()
340            .validate()
341            .is_err()
342        );
343
344        // Specifying a dependency name as a plain-name without a package is an error
345        assert!(
346            ComponentDependencies::deserialize(toml! {
347                "plain-name" = { version = "0.1.0" }
348            })
349            .unwrap()
350            .validate()
351            .is_err()
352        );
353
354        // Specifying an export to satisfy a package dependency name is an error
355        assert!(
356            ComponentDependencies::deserialize(toml! {
357                "foo:baz@0.1.0" = { path = "foo.wasm", export = "foo"}
358            })
359            .unwrap()
360            .validate()
361            .is_err()
362        );
363
364        // Two compatible versions of the same package is an error
365        assert!(
366            ComponentDependencies::deserialize(toml! {
367                "foo:baz@0.1.0" = "0.1.0"
368                "foo:bar@0.2.1" = "0.2.1"
369                "foo:bar@0.2.2" = "0.2.2"
370            })
371            .unwrap()
372            .validate()
373            .is_err()
374        );
375
376        // Two disjoint versions of the same package is ok
377        assert!(
378            ComponentDependencies::deserialize(toml! {
379                "foo:bar@0.1.0" = "0.1.0"
380                "foo:bar@0.2.0" = "0.2.0"

382            })
383            .unwrap()
384            .validate()
385            .is_ok()
386        );
387
388        // Unversioned and versioned dependencies of the same package is an error

397        );
398
399        // Two interfaces of two disjoint versions of a package is ok
400        assert!(
401            ComponentDependencies::deserialize(toml! {

404            })
405            .unwrap()
406            .validate()
407            .is_ok()
408        );
409
410        // A versioned interface and a different versioned package is ok
411        assert!(
412            ComponentDependencies::deserialize(toml! {
413                "foo:bar/baz@0.1.0" = "0.1.0"
414                "foo:bar@0.2.0" = "0.2.0"
415            })
416            .unwrap()
417            .validate()
418            .is_ok()
419        );
