66/// Mode of assembly of a Spin application into an OCI image
67#[derive(Copy, Clone)]
68enum AssemblyMode {
69    /// Assemble the application as one layer per component and one layer for
70    /// every static asset included with a given component
71    Simple,
72    /// Assemble the application as one layer per component and one compressed
73    /// archive layer containing all static assets included with a given component
74    Archive,
75}

77/// Indicates whether to compose the components of a Spin application when pushing an image.
78#[derive(Copy, Clone)]
79pub enum ComposeMode {
80    /// Compose components before pushing the image.
81    All,
82    /// Skip composing components before pushing the image.
83    Skip,
84}

130    /// Push a Spin application to an OCI registry and return the digest (or None
131    /// if the digest cannot be determined).
132    pub async fn push(
133        &mut self,
134        manifest_path: &Path,
135        profile: Option<&str>,
136        reference: impl AsRef<str>,
137        annotations: Option<BTreeMap<String, String>>,
138        infer_annotations: InferPredefinedAnnotations,
139        compose_mode: ComposeMode,
140    ) -> Result<Option<String>> {
141        let reference: Reference = reference
142            .as_ref()
143            .parse()
144            .with_context(|| format!("cannot parse reference {}", reference.as_ref()))?;
145        let auth = Self::auth(&reference).await?;
146        let working_dir = tempfile::tempdir()?;
147
148        // Create a locked application from the application manifest.
149        // TODO: We don't need an extra copy here for each asset to prepare the application.
150        // We should be able to use assets::collect instead when constructing the locked app.
151        let locked = spin_loader::from_file(
152            manifest_path,
153            FilesMountStrategy::Copy(working_dir.path().into()),
154            profile,
155            None,
156        )
157        .await?;
158
159        // Ensure that all Spin components specify valid wasm binaries for the `source`
160        // field, each dependency, and each trigger dependency.
161        for locked_component in &locked.components {
162            validate::ensure_wasms(locked_component).await?;
163        }
164
165        self.push_locked_core(
166            locked,
167            auth,
168            reference,
169            annotations,
170            infer_annotations,
171            compose_mode,
172        )
173        .await
174    }

203    /// Push a Spin application to an OCI registry and return the digest (or None
204    /// if the digest cannot be determined).
205    async fn push_locked_core(
206        &mut self,
207        locked: LockedApp,
208        auth: RegistryAuth,
209        reference: Reference,
210        annotations: Option<BTreeMap<String, String>>,
211        infer_annotations: InferPredefinedAnnotations,
212        compose_mode: ComposeMode,
213    ) -> Result<Option<String>> {
214        let mut locked_app = locked.clone();
215        let mut layers = self
216            .assemble_layers(&mut locked_app, AssemblyMode::Simple, compose_mode)
217            .await
218            .context("could not assemble layers for locked application")?;
219
220        // If SPIN_OCI_ARCHIVE_LAYERS_OPT is set *or* if layer count exceeds MAX_LAYER_COUNT-1,
221        // assemble archive layers instead. (An additional layer to represent the locked
222        // application config is added.)
223        if std::env::var(SPIN_OCI_ARCHIVE_LAYERS_OPT).is_ok() || layers.len() > MAX_LAYER_COUNT - 1
224        {
225            locked_app = locked.clone();
226            layers = self
227                .assemble_layers(&mut locked_app, AssemblyMode::Archive, compose_mode)
228                .await
229                .context("could not assemble archive layers for locked application")?;
230        }
231
232        let annotations = all_annotations(&locked_app, annotations, infer_annotations);
233
234        // Push layer for locked spin application config
235        let locked_config_layer = ImageLayer::new(
236            serde_json::to_vec(&locked_app).context("could not serialize locked config")?,
237            SPIN_APPLICATION_MEDIA_TYPE.to_string(),
238            None,
239        );
240        let config_layer_digest = locked_config_layer.sha256_digest().clone();
241        layers.push(locked_config_layer);
242
243        let mut labels = HashMap::new();
244        labels.insert(
245            "com.fermyon.spin.lockedAppDigest".to_string(),
246            config_layer_digest,
247        );
248        let cfg = oci_client::config::Config {
249            labels: Some(labels),
250            ..Default::default()
251        };
252
253        // Construct empty/default OCI config file. Data may be parsed according to
254        // the expected config structure per the image spec, so we want to ensure it conforms.
255        // (See https://github.com/opencontainers/image-spec/blob/main/config.md)
256        // TODO: Explore adding data applicable to the Spin app being published.
257        let oci_config_file = ConfigFile {
258            architecture: oci_client::config::Architecture::Wasm,
259            os: oci_client::config::Os::Other("Wasip1".to_string()),
260            // We need to ensure that the image config for different content is updated.
261            // Without referencing the digest of the locked application in the OCI image config,
262            // all Spin applications would get the same image config digest, resulting in the same
263            // image ID in container runtimes.
264            config: Some(cfg),
265            ..Default::default()
266        };
267        let oci_config =
268            oci_client::client::Config::oci_v1_from_config_file(oci_config_file, None)?;
269        let manifest = OciImageManifest::build(&layers, &oci_config, annotations);
270
271        let response = self
272            .oci
273            .push(&reference, &layers, oci_config, &auth, Some(manifest))
274            .await
275            .map(|push_response| push_response.manifest_url)
276            .context("cannot push Spin application")?;
277
278        tracing::info!("Pushed {:?}", response);
279
280        let digest = digest_from_url(&response);
281        Ok(digest)
282    }

284    /// Assemble ImageLayers for a locked application using the provided
285    /// AssemblyMode and return the resulting Vec<ImageLayer>.
286    async fn assemble_layers(
287        &mut self,
288        locked: &mut LockedApp,
289        assembly_mode: AssemblyMode,
290        compose_mode: ComposeMode,
291    ) -> Result<Vec<ImageLayer>> {
292        let (mut layers, components) = match compose_mode {
293            ComposeMode::All => {
294                self.assemble_layers_composed(assembly_mode, locked.clone())
295                    .await?
296            }
297            ComposeMode::Skip => {
298                self.assemble_layers_uncomposed(assembly_mode, locked.clone())
299                    .await?
300            }
301        };
302
303        locked.components = components;
304        locked.metadata.remove("origin");
305
306        // Deduplicate layers
307        layers = layers.into_iter().unique().collect();
308
309        Ok(layers)
310    }

312    async fn assemble_layers_uncomposed(
313        &mut self,
314        assembly_mode: AssemblyMode,
315        locked: LockedApp,
316    ) -> Result<(Vec<ImageLayer>, Vec<LockedComponent>)> {
317        let mut components = Vec::new();
318        let mut layers = Vec::new();
319
320        for mut c in locked.components {
321            // Add the wasm module for the component as layers.
322            let source = c
323                .source
324                .content
325                .source
326                .as_ref()
327                .context("component loaded from disk should contain a file source")?;
328
329            let source = parse_file_url(source.as_str())?;
330            let layer = Self::wasm_layer(&source).await?;
331
332            // Update the module source with the content ref of the layer.
333            c.source.content = self.content_ref_for_layer(&layer);
334
335            layers.push(layer);
336
337            for dep in c.dependencies.values_mut() {
338                self.assemble_dependency_layer(dep, &mut layers).await?;
339            }
340
341            for deps in c.trigger_dependencies.values_mut() {
342                for dep in deps {
343                    self.assemble_dependency_layer(dep, &mut layers).await?;
344                }
345            }
346
347            c.files = self
348                .assemble_content_layers(assembly_mode, &mut layers, c.files.as_slice())
349                .await?;
350            components.push(c);
351        }
352
353        Ok((layers, components))
354    }

379    async fn assemble_layers_composed(
380        &mut self,
381        assembly_mode: AssemblyMode,
382        locked: LockedApp,
383    ) -> Result<(Vec<ImageLayer>, Vec<LockedComponent>)> {
384        let mut components = Vec::new();
385        let mut layers = Vec::new();
386
387        let temp_dir =
388            tempfile::tempdir().context("unable to create tempdir for precomposition")?;
389        let working_dir = temp_dir.path();
390        let locked_url = write_locked_app(&locked, working_dir)
391            .await
392            .context("unable to write locked app for precomposition")?;
393
394        for mut c in locked.components {
395            let trigger_deps = &c.trigger_dependencies;
396
397            let composed = if trigger_deps.is_empty() {
398                spin_compose::compose(&ComponentSourceLoaderFs, &c, async |a| Ok(a)).await?
399            } else {
400                // There are trigger deps (e.g. middleware): we need to hand off to the trigger
401                // to do the composition.
402                precompose_using_trigger(&c, &locked_url, working_dir).await?
403            };
404
405            let layer = ImageLayer::new(composed, WASM_LAYER_MEDIA_TYPE.to_string(), None);
406            c.source.content = self.content_ref_for_layer(&layer);
407            c.dependencies.clear();
408            c.trigger_dependencies.clear();
409            layers.push(layer);
410
411            c.files = self
412                .assemble_content_layers(assembly_mode, &mut layers, c.files.as_slice())
413                .await?;
414            components.push(c);
415        }
416
417        // Copied from `spin up`
418        async fn write_locked_app(
419            locked_app: &LockedApp,
420            working_dir: &Path,
421        ) -> Result<String, anyhow::Error> {
422            let locked_path = working_dir.join("spin.lock");
423            let locked_app_contents =
424                serde_json::to_vec_pretty(&locked_app).context("failed to serialize locked app")?;
425            tokio::fs::write(&locked_path, locked_app_contents)
426                .await
427                .with_context(|| format!("failed to write {}", quoted_path(&locked_path)))?;
428            let locked_url = Url::from_file_path(&locked_path)
429                .map_err(|_| {
430                    anyhow::anyhow!("cannot convert to file URL: {}", quoted_path(&locked_path))
431                })?
432                .to_string();
433
434            Ok(locked_url)
435        }
436
437        Ok((layers, components))
438    }

440    async fn assemble_content_layers(
441        &mut self,
442        assembly_mode: AssemblyMode,
443        layers: &mut Vec<ImageLayer>,
444        contents: &[ContentPath],
445    ) -> Result<Vec<ContentPath>> {
446        let mut files = Vec::new();
447        for f in contents {
448            let source = f
449                .content
450                .source
451                .as_ref()
452                .context("file mount loaded from disk should contain a file source")?;
453            let source = parse_file_url(source.as_str())?;
454
455            match assembly_mode {
456                AssemblyMode::Archive => self
457                    .push_archive_layer(&source, &mut files, layers)
458                    .await
459                    .context(format!(
460                        "cannot push archive layer for source {}",
461                        quoted_path(&source)
462                    ))?,
463                AssemblyMode::Simple => self
464                    .push_file_layers(&source, &mut files, layers)
465                    .await
466                    .context(format!(
467                        "cannot push file layers for source {}",
468                        quoted_path(&source)
469                    ))?,
470            }
471        }
472        Ok(files)
473    }

475    /// Archive all of the files recursively under the source directory
476    /// and push as a compressed archive layer
477    async fn push_archive_layer(
478        &mut self,
479        source: &PathBuf,
480        files: &mut Vec<ContentPath>,
481        layers: &mut Vec<ImageLayer>,
482    ) -> Result<()> {
483        // Add all archived file entries to the locked app manifest
484        for entry in WalkDir::new(source) {
485            let entry = entry?;
486            if !entry.file_type().is_file() {
487                continue;
488            }
489            // Can unwrap because we got to 'entry' from walking 'source'
490            let rel_path = entry.path().strip_prefix(source).unwrap();
491            tracing::trace!("Adding asset {rel_path:?} to component files list");
492            // Add content/path to the locked component files list
493            let layer = Self::data_layer(entry.path(), DATA_MEDIATYPE.to_string()).await?;
494            let content = self.content_ref_for_layer(&layer);
495            files.push(ContentPath {
496                content,
497                path: rel_path.into(),
498            });
499        }
500
501        // Only add the archive layer to the OCI manifest
502        tracing::trace!("Adding archive layer for all files in source {:?}", &source);
503        let working_dir = tempfile::tempdir()?;
504        let archive_path = crate::utils::archive(source, &working_dir.keep())
505            .await
506            .context(format!(
507                "Unable to create compressed archive for source {source:?}"
508            ))?;
509        let layer = Self::data_layer(archive_path.as_path(), ARCHIVE_MEDIATYPE.to_string()).await?;
510        layers.push(layer);
511        Ok(())
512    }

514    /// Recursively traverse the source directory and add layers for each file.
515    async fn push_file_layers(
516        &mut self,
517        source: &PathBuf,
518        files: &mut Vec<ContentPath>,
519        layers: &mut Vec<ImageLayer>,
520    ) -> Result<()> {
521        // Traverse each mount directory, add all static assets as layers, then update the
522        // locked application file with the file digest.
523        tracing::trace!("Adding new layer per file under source {:?}", source);
524        for entry in WalkDir::new(source) {
525            let entry = entry?;
526            if !entry.file_type().is_file() {
527                continue;
528            }
529            // Can unwrap because we got to 'entry' from walking 'source'
530            let rel_path = entry.path().strip_prefix(source).unwrap();
531            // Paths must be in portable (forward slash) format in the registry,
532            // so that they can be placed correctly on any host system
533            let rel_path = portable_path(rel_path);
534
535            tracing::trace!("Adding new layer for asset {rel_path:?}");
536            // Construct and push layer, adding its digest to the locked component files Vec
537            let layer = Self::data_layer(entry.path(), DATA_MEDIATYPE.to_string()).await?;
538            let content = self.content_ref_for_layer(&layer);
539            let content_inline = content.inline.is_some();
540            files.push(ContentPath {
541                content,
542                path: rel_path,
543            });
544            // As a workaround for OCI implementations that don't support very small blobs,
545            // don't push very small content that has been inlined into the manifest:
546            // https://github.com/distribution/distribution/discussions/4029
547            let skip_layer = content_inline;
548            if !skip_layer {
549                layers.push(layer);
550            }
551        }
552        Ok(())
553    }

555    /// Pull a Spin application from an OCI registry.
556    pub async fn pull(&mut self, reference: &str) -> Result<OciImageManifest> {
557        let reference: Reference = reference.parse().context("cannot parse reference")?;
558        let auth = Self::auth(&reference).await?;
559
560        // Pull the manifest from the registry.
561        let (manifest, digest) = self.oci.pull_image_manifest(&reference, &auth).await?;
562
563        let manifest_json = serde_json::to_string(&manifest)?;
564        tracing::debug!("Pulled manifest: {}", manifest_json);
565
566        // Write the manifest in `<cache_root>/registry/oci/manifests/repository:<tag_or_latest>/manifest.json`
567        let m = self.manifest_path(&reference.to_string()).await?;
568        fs::write(&m, &manifest_json).await?;
569
570        // Older published Spin apps feature the locked app config *as* the OCI manifest config layer,
571        // while newer versions publish the locked app config as a generic layer alongside others.
572        // Assume that these bytes may represent the locked app config and write it as such.
573        let mut cfg_bytes = Vec::new();
574        self.oci
575            .pull_blob(&reference, &manifest.config, &mut cfg_bytes)
576            .await?;
577        self.write_locked_app_config(&reference.to_string(), &cfg_bytes)
578            .await
579            .context("unable to write locked app config to cache")?;
580
581        // If a layer is a Wasm module, write it in the Wasm directory.
582        // Otherwise, write it in the data directory (after unpacking if archive layer)
583        stream::iter(&manifest.layers)
584            .map(|layer| {
585                let this = &self;
586                let reference = reference.clone();
587                async move {
588                    // Skip pulling if the digest already exists in the wasm or data directories.
589                    if this.cache.wasm_file(&layer.digest).is_ok()
590                        || this.cache.data_file(&layer.digest).is_ok()
591                    {
592                        tracing::debug!("Layer {} already exists in cache", &layer.digest);
593                        return anyhow::Ok(());
594                    }
595
596                    tracing::debug!("Pulling layer {}", &layer.digest);
597                    let mut bytes = Vec::with_capacity(layer.size.try_into()?);
598                    this.oci.pull_blob(&reference, layer, &mut bytes).await?;
599                    match layer.media_type.as_str() {
600                        SPIN_APPLICATION_MEDIA_TYPE => {
601                            this.write_locked_app_config(&reference.to_string(), &bytes)
602                                .await
603                                .with_context(|| "unable to write locked app config to cache")?;
604                        }
605                        WASM_LAYER_MEDIA_TYPE | WASM_LAYER_MEDIA_TYPE_WKG => {
606                            this.cache.write_wasm(&bytes, &layer.digest).await?;
607                        }
608                        ARCHIVE_MEDIATYPE => {
609                            unpack_archive_layer(&this.cache, &bytes, &layer.digest).await?;
610                        }
611                        _ => {
612                            this.cache.write_data(&bytes, &layer.digest).await?;
613                        }
614                    }
615                    Ok(())
616                }
617            })
618            .buffer_unordered(MAX_PARALLEL_PULL)
619            .try_for_each(future::ok)
620            .await?;
621        tracing::info!("Pulled {}@{}", reference, digest);
622
623        Ok(manifest)
624    }

626    /// Get the file path to an OCI manifest given a reference.
627    /// If the directory for the manifest does not exist, this will create it.
628    async fn manifest_path(&self, reference: impl AsRef<str>) -> Result<PathBuf> {
629        let reference: Reference = reference
630            .as_ref()
631            .parse()
632            .context("cannot parse OCI reference")?;
633        let p = self
634            .cache
635            .manifests_dir()
636            .join(fs_safe_segment(reference.registry()))
637            .join(reference.repository())
638            .join(reference.tag().unwrap_or(LATEST_TAG));
639
640        if !p.is_dir() {
641            fs::create_dir_all(&p).await.with_context(|| {
642                format!("cannot create directory {} for OCI manifest", p.display())
643            })?;
644        }
645
646        Ok(p.join(MANIFEST_FILE))
647    }

649    /// Get the file path to the OCI configuration object given a reference.
650    pub async fn lockfile_path(&self, reference: impl AsRef<str>) -> Result<PathBuf> {
651        let reference: Reference = reference

719    /// Save a credential set containing the registry username and password.
720    pub async fn login(
721        server: impl AsRef<str>,
722        username: impl AsRef<str>,
723        password: impl AsRef<str>,
724    ) -> Result<()> {
725        let registry = registry_from_input(server);
726
727        // First, validate the credentials. If a user accidentally enters a wrong credential set, this
728        // can catch the issue early rather than getting an error at the first operation that needs
729        // to use the credentials (first time they do a push/pull/up).
730        Self::validate_credentials(&registry, &username, &password).await?;
731
732        // Save an encoded representation of the credential set in the local configuration file.
733        let mut auth = AuthConfig::load_default().await?;
734        auth.insert(registry, username, password)?;
735        auth.save_default().await
736    }

748    /// Validate the credentials by attempting to send an authenticated request to the registry.
749    async fn validate_credentials(
750        server: impl AsRef<str>,
751        username: impl AsRef<str>,
752        password: impl AsRef<str>,
753    ) -> Result<()> {
754        let client = docker_registry::v2::Client::configure()
755            .registry(server.as_ref())
756            .insecure_registry(false)
757            .username(Some(username.as_ref().into()))
758            .password(Some(password.as_ref().into()))
759            .build()
760            .context("cannot create client to send authentication request to the registry")?;
761
762        match client
763            // We don't need to configure any scopes, we are only testing that the credentials are
764            // valid for the intended registry.
765            .authenticate(&[""])
766            .await
767        {
768            Ok(_) => Ok(()),
769            Err(e) => bail!(format!(
770                "cannot authenticate as {} to registry {}: {}",
771                username.as_ref(),
772                server.as_ref(),
773                e
774            )),
775        }
776    }

778    /// Construct the registry authentication based on the reference.
779    async fn auth(reference: &Reference) -> Result<RegistryAuth> {
780        let server = reference
781            .resolve_registry()
782            .strip_suffix('/')
783            .unwrap_or_else(|| reference.resolve_registry());
784
785        match AuthConfig::get_auth_from_default(server).await {
786            Ok(c) => Ok(c),
787            Err(_) => match docker_credential::get_credential(server) {
788                Err(e) => {
789                    tracing::trace!(
790                        "Cannot retrieve credentials from Docker, attempting to use anonymous auth: {}",
791                        e
792                    );
793                    Ok(RegistryAuth::Anonymous)
794                }
795
796                Ok(DockerCredential::UsernamePassword(username, password)) => {
797                    tracing::trace!("Found Docker credentials");
798                    Ok(RegistryAuth::Basic(username, password))
799                }
800                Ok(DockerCredential::IdentityToken(_)) => {
801                    tracing::trace!(
802                        "Cannot use contents of Docker config, identity token not supported. Using anonymous auth"
803                    );
804                    Ok(RegistryAuth::Anonymous)
805                }
806            },
807        }
808    }

826/// Unpack contents of the provided archive layer, represented by bytes and its
827/// corresponding digest, into the provided cache.
828/// A temporary staging directory is created via tempfile::tempdir() to store
829/// the unpacked contents prior to writing to the cache.
830pub async fn unpack_archive_layer(
831    cache: &Cache,
832    bytes: impl AsRef<[u8]>,
833    digest: impl AsRef<str>,
834) -> Result<()> {
835    // Write archive layer to cache as usual
836    cache.write_data(&bytes, &digest).await?;
837
838    // Unpack archive into a staging dir
839    let path = cache
840        .data_file(&digest)
841        .context("unable to read archive layer from cache")?;
842    let staging_dir = tempfile::tempdir()?;
843    crate::utils::unarchive(path.as_ref(), staging_dir.path()).await?;
844
845    // Traverse unpacked contents and if a file, write to cache by digest
846    // (if it doesn't already exist)
847    for entry in WalkDir::new(staging_dir.path()) {
848        let entry = entry?;
849        if entry.file_type().is_file() && !entry.file_type().is_dir() {
850            let bytes = tokio::fs::read(entry.path()).await?;
851            let digest = format!("sha256:{}", sha256::hex_digest_from_bytes(&bytes));
852            if cache.data_file(&digest).is_ok() {
853                tracing::debug!(
854                    "Skipping unpacked asset {:?}; file already exists",
855                    entry.path()
856                );
857            } else {
858                tracing::debug!("Adding unpacked asset {:?} to cache", entry.path());
859                cache.write_data(bytes, &digest).await?;
860            }
861        }
862    }
863    Ok(())
864}

866fn digest_from_url(manifest_url: &str) -> Option<String> {
867    // The URL is in the form "https://host/v2/refname/manifests/sha256:..."
868    let manifest_url = Url::parse(manifest_url).ok()?;
869    let mut segments = manifest_url.path_segments()?;
870    let last = segments.next_back()?;
871    if last.contains(':') {
872        Some(last.to_owned())
873    } else {
874        None
875    }
876}

878fn registry_from_input(server: impl AsRef<str>) -> String {
879    // We want to allow a user to login to both https://ghcr.io and ghcr.io.
880    let server = server.as_ref();
881    let server = match server.parse::<Url>() {
882        Ok(url) => url.host_str().unwrap_or(server).to_string(),
883        Err(_) => server.to_string(),
884    };
885    // DockerHub is commonly referenced as 'docker.io' but needs to be 'index.docker.io'
886    match server.as_str() {
887        "docker.io" => "index.docker.io".to_string(),
888        _ => server,
889    }
890}

892fn all_annotations(
893    locked_app: &LockedApp,
894    explicit: Option<BTreeMap<String, String>>,
895    predefined: InferPredefinedAnnotations,
896) -> Option<BTreeMap<String, String>> {
897    use spin_locked_app::{APP_DESCRIPTION_KEY, APP_NAME_KEY, APP_VERSION_KEY, MetadataKey};
898    const APP_AUTHORS_KEY: MetadataKey<Vec<String>> = MetadataKey::new("authors");
899
900    if predefined == InferPredefinedAnnotations::None {
901        return explicit;
902    }
903
904    // We will always, at minimum, have a `created` annotation, so if we don't already have an
905    // anootations collection then we may as well create one now...
906    let mut current = explicit.unwrap_or_default();
907
908    let authors = locked_app
909        .get_metadata(APP_AUTHORS_KEY)
910        .unwrap_or_default()
911        .unwrap_or_default();
912    if !authors.is_empty() {
913        let authors = authors.join(", ");
914        add_inferred(
915            &mut current,
916            oci_client::annotations::ORG_OPENCONTAINERS_IMAGE_AUTHORS,
917            Some(authors),
918        );
919    }
920
921    let name = locked_app.get_metadata(APP_NAME_KEY).unwrap_or_default();
922    add_inferred(
923        &mut current,
924        oci_client::annotations::ORG_OPENCONTAINERS_IMAGE_TITLE,
925        name,
926    );
927
928    let description = locked_app
929        .get_metadata(APP_DESCRIPTION_KEY)
930        .unwrap_or_default();
931    add_inferred(
932        &mut current,
933        oci_client::annotations::ORG_OPENCONTAINERS_IMAGE_DESCRIPTION,
934        description,
935    );
936
937    let version = locked_app.get_metadata(APP_VERSION_KEY).unwrap_or_default();
938    add_inferred(
939        &mut current,
940        oci_client::annotations::ORG_OPENCONTAINERS_IMAGE_VERSION,
941        version,
942    );
943
944    let created = chrono::Utc::now().to_rfc3339();
945    add_inferred(
946        &mut current,
947        oci_client::annotations::ORG_OPENCONTAINERS_IMAGE_CREATED,
948        Some(created),
949    );
950
951    Some(current)
952}
953
954fn add_inferred(map: &mut BTreeMap<String, String>, key: &str, value: Option<String>) {
955    if let Some(value) = value
956        && let std::collections::btree_map::Entry::Vacant(e) = map.entry(key.to_string())
957    {
958        e.insert(value);
959    }
960}

963const SPIN_WORKING_DIR: &str = "SPIN_WORKING_DIR";
964
965async fn precompose_using_trigger(
966    c: &LockedComponent,
967    locked_url: &str,
968    working_dir: &Path,
969) -> Result<Vec<u8>, spin_compose::ComposeError> {
970    use spin_compose::ComposeError;
971
972    let Some(resolve_extras_using) = c
973        .metadata
974        .get("resolve-trigger-dependencies-using")
975        .and_then(|v| v.as_str())
976    else {
977        return spin_compose::compose(&ComponentSourceLoaderFs, c, async |a| Ok(a)).await;
978    };
979
980    let resolver_subcmd = match resolve_extras_using {
981        "http" | "redis" => vec!["trigger".into(), resolve_extras_using.into()],
982        _ => vec![format!("trigger-{resolve_extras_using}")],
983    };
984
985    let mut cmd = tokio::process::Command::new(std::env::current_exe().unwrap());
986    cmd.args(resolver_subcmd)
987        .args(["--precompose-only", "--precompose-component-id"])
988        .arg(&c.id)
989        .stdout(std::process::Stdio::piped())
990        .stderr(std::process::Stdio::inherit())
991        .env("SPIN_PLUGINS_SUPPRESS_COMPATIBILITY_WARNINGS", "1")
992        .env(SPIN_LOCKED_URL, locked_url)
993        .env(SPIN_WORKING_DIR, working_dir);
994
995    let child = cmd
996        .spawn()
997        .map_err(|e| ComposeError::PrepareError(e.into()))?;
998
999    let trigger_out = child
1000        .wait_with_output()
1001        .await
1002        .map_err(|e| ComposeError::PrepareError(e.into()))?;
1003
1004    if !trigger_out.status.success() {
1005        return Err(ComposeError::PrepareError(anyhow::anyhow!(
1006            "unable to compose additional components for {} using `{}`",
1007            c.id,
1008            resolve_extras_using
1009        )));
1010    }
1011
1012    let composed = trigger_out.stdout;
1013    Ok(composed)
1014}

1482    /// Regression test: when composition is skipped, a component's trigger dependencies
1483    /// (e.g. HTTP middleware) must be pushed as layers in their own right. Previously
1484    /// they were published still pointing at the pushing machine's filesystem, which made
1485    /// the resulting image unusable anywhere else.
1486    #[tokio::test]
1487    async fn skipping_composition_pushes_trigger_dependency_layers() {
1488        use tokio::io::AsyncWriteExt;
1489
1490        let working_dir = tempfile::tempdir().unwrap();
1491
1492        for name in ["component1.wasm", "middleware.wasm"] {
1493            let mut f = tokio::fs::File::create(working_dir.path().join(name))
1494                .await
1495                .expect("should create wasm file");
1496            f.write_all(name.as_bytes())
1497                .await
1498                .expect("should write wasm contents");
1499        }
1500
1501        let mut locked = LockedApp {
1502            spin_lock_version: Default::default(),
1503            components: from_json!([{
1504                "id": "component1",
1505                "source": {
1506                    "content_type": "application/wasm",
1507                    "source": file_url(working_dir.path().join("component1.wasm")),
1508                    "digest": "digest",
1509                },
1510                "trigger_dependencies": {
1511                    "middleware": [{
1512                        "source": {
1513                            "content_type": "application/wasm",
1514                            "source": file_url(working_dir.path().join("middleware.wasm")),
1515                            "digest": "digest",
1516                        },
1517                        "export": null,
1518                    }]
1519                }
1520            }]),
1521            triggers: Default::default(),
1522            metadata: Default::default(),
1523            variables: Default::default(),
1524            must_understand: Default::default(),
1525            host_requirements: Default::default(),
1526        };
1527
1528        let mut client = Client::new(false, Some(working_dir.path().to_path_buf()))
1529            .await
1530            .expect("should create new client");
1531        client.opts = ClientOpts {
1532            content_ref_inline_max_size: 0,
1533        };
1534
1535        let layers = client
1536            .assemble_layers(&mut locked, AssemblyMode::Simple, ComposeMode::Skip)
1537            .await
1538            .expect("should assemble layers");
1539
1540        assert_eq!(
1541            2,
1542            layers.len(),
1543            "expected one layer for the component and one for its trigger dependency"
1544        );
1545
1546        let dep = &locked.components[0].trigger_dependencies["middleware"][0];
1547        assert!(
1548            dep.source.content.source.is_none(),
1549            "trigger dependency should no longer refer to a local file, but got {:?}",
1550            dep.source.content.source
1551        );
1552        assert!(
1553            dep.source.content.digest.is_some(),
1554            "trigger dependency should refer to its pushed layer by digest"
1555        );
1556    }
