8/// Creates a temporary file in the directory of `path`, to be persisted over it.
9/// On Unix the file is created with mode 0666 less the umask, as `std::fs::write`
10/// would create it, rather than tempfile's default of 0600.
11pub fn temp_file_beside(path: &Path) -> anyhow::Result<tempfile::NamedTempFile> {
12    let dir = path
13        .parent()
14        .with_context(|| format!("{} has no parent directory", path.display()))?;
15    let mut builder = tempfile::Builder::new();
16    #[cfg(unix)]
17    {
18        use std::os::unix::fs::PermissionsExt;
19        builder.permissions(std::fs::Permissions::from_mode(0o666));
20    }
21    builder
22        .tempfile_in(dir)
23        .with_context(|| format!("cannot create temporary file in {}", dir.display()))
24}

26/// Writes `bytes` to `path` through a temporary file in the same directory
27/// that is then renamed over `path`, so a concurrent reader sees either the
28/// previous content or the new content, never a partial file.
29pub async fn write_file_atomically_stream(
30    path: &Path,
31    bytes: impl futures::Stream<Item = Result<bytes::Bytes, std::io::Error>> + Unpin,
32) -> anyhow::Result<()> {
33    let tmp = temp_file_beside(path)?;
34    let mut bytes = tokio_util::io::StreamReader::new(bytes);
35    let mut f = tokio::fs::File::create(tmp.path()).await?;
36    tokio::io::copy(&mut bytes, &mut f).await?;
37    tmp.persist(path)
38        .with_context(|| format!("cannot write {}", quoted_path(path)))?;
39    Ok(())
40}
