6/// Encapsulates checking of a PostgreSQL address/connection string against
7/// an allow-list.
8///
9/// This is broken out as a distinct object to allow it to be synchronously retrieved
10/// within a P3 Accessor block and then asynchronously queried outside the block.
11#[derive(Clone)]
12pub(crate) struct AllowedHostChecker {
13    allowed_hosts: Arc<OutboundAllowedHosts>,
14}
15
16impl AllowedHostChecker {
17    pub fn new(allowed_hosts: OutboundAllowedHosts) -> Self {
18        Self {
19            allowed_hosts: Arc::new(allowed_hosts),
20        }
21    }
22
23    #[allow(clippy::result_large_err)]
24    pub async fn ensure_address_allowed(&self, address: &str) -> Result<(), v4::Error> {
25        fn conn_failed(message: impl Into<String>) -> v4::Error {
26            v4::Error::ConnectionFailed(message.into())
27        }
28        fn err_other(err: anyhow::Error) -> v4::Error {
29            v4::Error::Other(err.to_string())
30        }
31
32        let config = address
33            .parse::<tokio_postgres::Config>()
34            .map_err(|e| conn_failed(e.to_string()))?;
35
36        if config.get_hosts().is_empty() {
37            return Err(v4::Error::Other(
38                "No host in Postgres connection string".into(),
39            ));
40        }
41
42        for (i, host) in config.get_hosts().iter().enumerate() {
43            match host {
44                tokio_postgres::config::Host::Tcp(address) => {
45                    let ports = config.get_ports();
46                    // The port we use is either:
47                    // * The port at the same index as the host
48                    // * The first port if there is only one port
49                    let port = ports.get(i).or_else(|| {
50                        if ports.len() == 1 {
51                            ports.first()
52                        } else {
53                            None
54                        }
55                    });
56                    let port_str = port.map(|p| format!(":{p}")).unwrap_or_default();
57
58                    let hostaddrs = config.get_hostaddrs();
59                    let hostaddr = hostaddrs.get(i).or_else(|| {
60                        if hostaddrs.len() == 1 {
61                            hostaddrs.first()
62                        } else {
63                            None
64                        }
65                    });
66                    if let Some(hostaddr) = hostaddr {
67                        let hostaddr_url = format!("{hostaddr}{port_str}");
68                        if !self
69                            .allowed_hosts
70                            .check_url(&hostaddr_url, "postgres")
71                            .await
72                            .map_err(err_other)?
73                        {
74                            return Err(conn_failed(format!(
75                                "Postgres hostaddr {hostaddr} is not permitted"
76                            )));
77                        }
78                    }
79
80                    let url = format!("{address}{port_str}");
81                    if !self
82                        .allowed_hosts
83                        .check_url(&url, "postgres")
84                        .await
85                        .map_err(err_other)?
86                    {
87                        return Err(conn_failed(format!(
88                            "address postgres://{url} is not permitted"
89                        )));
90                    }
91                }
92                #[cfg(unix)]
93                tokio_postgres::config::Host::Unix(_) => {
94                    return Err(conn_failed("Unix sockets are not supported on WebAssembly"));
95                }
96            }
97        }
98
99        Ok(())
100    }
101}

103#[cfg(test)]
104mod test {
105    use super::*;
106    use futures_util::future::FutureExt;
107    use spin_factor_outbound_networking::config::allowed_hosts::{
108        AllowedHostConfig, AllowedHostsConfig,

111    fn make_checker_allowing(hosts: &[&str]) -> AllowedHostChecker {
112        let allowed_host_configs = hosts
113            .iter()
114            .map(|h| AllowedHostConfig::parse(*h).unwrap())
115            .collect();
116        let allowed_hosts = Arc::new(AllowedHostsConfig::SpecificHosts(allowed_host_configs));
117        let allowed = OutboundAllowedHosts::new(
118            (async move { Ok(allowed_hosts) }).boxed().shared(),
119            None,
120            false,
121        );
122        AllowedHostChecker {
123            allowed_hosts: Arc::new(allowed),
124        }
125    }
126
127    // This is permitted by Postgres but we are disallowing it until someone needs it
128    // because it makes the validation logic more complicated!
129    #[tokio::test]
130    async fn hostaddr_only_is_disallowed() {
131        let checker = make_checker_allowing(&["postgres://localhost"]);
132        let address = "hostaddr=192.168.45.67";
133        checker
134            .ensure_address_allowed(address)
135            .await
136            .expect_err("address should not have been allowed");
137    }
