12/// OciLoader loads an OCI app in preparation for running with Spin.
13pub struct OciLoader {
14    working_dir: PathBuf,
15}

17/// The artifact loaded by the `OciLoader`
18pub enum ExecutableArtifact {
19    /// The OCI reference contained a Spin application
20    Application(LockedApp),
21    /// The OCI reference contained a Wasm package
22    Package(PathBuf),
23}

25impl OciLoader {
26    /// Creates a new OciLoader which builds temporary mount directory(s) in
27    /// the given working_dir.
28    pub fn new(working_dir: impl Into<PathBuf>) -> Self {
29        let working_dir = working_dir.into();
30        Self { working_dir }

33    /// Pulls and loads an OCI Artifact and returns a LockedApp with the given OCI client and reference
34    pub async fn load_app(
35        &self,
36        client: &mut Client,
37        reference: &str,
38    ) -> Result<ExecutableArtifact> {
39        // Fetch app
40        let manifest = client.pull(reference).await.with_context(|| {
41            format!("cannot pull Spin application from registry reference {reference:?}")
42        })?;
43
44        // Read locked app
45        let lockfile_path = client
46            .lockfile_path(&reference)
47            .await
48            .context("cannot get path to spin.lock")?;
49        self.load_from_cache(manifest, lockfile_path, reference, &client.cache)
50            .await
51    }
52
53    /// Loads an OCI Artifact from the given cache and returns a LockedApp with the given reference
54    pub async fn load_from_cache(
55        &self,
56        manifest: oci_client::manifest::OciImageManifest,
57        lockfile_path: PathBuf,
58        reference: &str,
59        cache: &Cache,
60    ) -> std::result::Result<ExecutableArtifact, anyhow::Error> {
61        let locked_content = tokio::fs::read(&lockfile_path)
62            .await
63            .with_context(|| format!("failed to read from {}", quoted_path(&lockfile_path)))?;
64        let locked_json: serde_json::Value = serde_json::from_slice(&locked_content)
65            .with_context(|| format!("OCI config {} is not JSON", quoted_path(&lockfile_path)))?;
66
67        if locked_json.get("spin_lock_version").is_some() {
68            let mut locked_app = LockedApp::from_json(&locked_content).with_context(|| {
69                format!(
70                    "failed to decode locked app from {}",
71                    quoted_path(&lockfile_path)
72                )
73            })?;
74
75            // Update origin metadata
76            let resolved_reference = Reference::try_from(reference).context("invalid reference")?;
77            let origin_uri = format!("{ORIGIN_URL_SCHEME}:{resolved_reference}");
78            locked_app
79                .metadata
80                .insert("origin".to_string(), origin_uri.into());
81
82            for component in &mut locked_app.components {
83                self.resolve_component_content_refs(component, cache)
84                    .await
85                    .with_context(|| {
86                        format!("failed to resolve content for component {:?}", component.id)
87                    })?;
88            }
89            Ok(ExecutableArtifact::Application(locked_app))
90        } else {
91            if manifest.layers.len() != 1 {
92                anyhow::bail!(
93                    "expected single layer in OCI package, found {} layers",
94                    manifest.layers.len()
95                );
96            }
97            let layer = &manifest.layers[0]; // guaranteed safe by previous check
98            let wasm_path = cache.wasm_path(&layer.digest);
99            Ok(ExecutableArtifact::Package(wasm_path))
100        }
101    }

103    /// Resolves all digest references in the component (including source, dependencies,
104    /// and asset files). Wasm sources are replaced with cache paths. Asset files are
105    /// collected to a mount path, and the component `files` amended to reflect that.
106    ///
107    /// This function assumes that:
108    ///
109    /// 1. All resolvable items have digest or inline sources.
110    /// 2. All digest-addressed content is already in the cache.
111    ///
112    /// If either of these is not true, the function errors.
113    pub async fn resolve_component_content_refs(
114        &self,
115        component: &mut LockedComponent,
116        cache: &Cache,
117    ) -> Result<()> {
118        ensure!(
119            is_safe_to_join(&component.id),
120            "invalid component id {}",
121            component.id
122        );
123
124        // Update wasm content path
125        let wasm_digest = content_digest(&component.source.content)?;
126        let wasm_path = cache.wasm_file(wasm_digest)?;
127        component.source.content = content_ref(wasm_path)?;
128
129        for dep in &mut component.dependencies.values_mut() {
130            let dep_wasm_digest = content_digest(&dep.source.content)?;
131            let dep_wasm_path = cache.wasm_file(dep_wasm_digest)?;
132            dep.source.content = content_ref(dep_wasm_path)?;
133        }
134
135        // Applications pushed without composition carry their trigger dependencies
136        // (e.g. HTTP middleware) as layers of their own, so those need resolving too.
137        for deps in component.trigger_dependencies.values_mut() {
138            for dep in deps {
139                let dep_wasm_digest = content_digest(&dep.source.content)?;
140                let dep_wasm_path = cache.wasm_file(dep_wasm_digest)?;
141                dep.source.content = content_ref(dep_wasm_path)?;
142            }
143        }
144
145        if !component.files.is_empty() {
146            let mount_dir = self.working_dir.join("assets").join(&component.id);
147            for file in &mut component.files {
148                ensure!(is_safe_to_join(&file.path), "invalid file mount {file:?}");
149                let mount_path = mount_dir.join(&file.path);
150
151                // Create parent directory
152                let mount_parent = mount_path
153                    .parent()
154                    .with_context(|| format!("invalid mount path {mount_path:?}"))?;
155                tokio::fs::create_dir_all(mount_parent)
156                    .await
157                    .with_context(|| {
158                        format!("failed to create temporary mount path {mount_path:?}")
159                    })?;
160
161                if let Some(content_bytes) = file.content.inline.as_deref() {
162                    // Write inline content to disk
163                    tokio::fs::write(&mount_path, content_bytes)
164                        .await
165                        .with_context(|| {
166                            format!("failed to write inline content to {mount_path:?}")
167                        })?;
168                } else {
169                    // Copy content
170                    let digest = content_digest(&file.content)?;
171                    let content_path = cache.data_file(digest)?;
172                    // TODO: parallelize
173                    tokio::fs::copy(&content_path, &mount_path)
174                        .await
175                        .with_context(|| {
176                            format!(
177                                "failed to copy {}->{mount_path:?}",
178                                quoted_path(&content_path)
179                            )
180                        })?;
181                }
182            }
183
184            component.files = vec![ContentPath {
185                content: content_ref(mount_dir)?,
186                path: "/".into(),
187            }]
188        }
189
190        Ok(())
191    }
192}

194fn content_digest(content_ref: &ContentRef) -> Result<&str> {
195    content_ref
196        .digest
197        .as_deref()
198        .with_context(|| format!("content missing expected digest: {content_ref:?}"))
199}
200
201fn content_ref(path: impl AsRef<Path>) -> Result<ContentRef> {
202    let path = std::fs::canonicalize(path)?;
203    let url = Url::from_file_path(path).map_err(|_| anyhow!("couldn't build file URL"))?;
204    Ok(ContentRef {
205        source: Some(url.to_string()),
206        ..Default::default()
207    })
208}
209
210fn is_safe_to_join(path: impl AsRef<Path>) -> bool {
211    // This could be loosened, but currently should always be true
212    path.as_ref()
213        .components()
214        .all(|c| matches!(c, std::path::Component::Normal(_)))
215}

221    async fn mount_test_component_assets(
222        component_id: &str,
223        working_dir: &Path,
224        cache_dir: &Path,
225    ) -> anyhow::Result<()> {
226        let loader = OciLoader::new(working_dir);
227        let cache = Cache::new(Some(cache_dir.into())).await?;
228
229        let fake_wasm = vec![1, 2, 3, 4];
230        let fake_wasm_digest = spin_common::sha256::hex_digest_from_bytes(&fake_wasm);
231        cache.write_wasm(&fake_wasm, &fake_wasm_digest).await?;
232
233        let fake_file = "some content".into();
234        let fake_file_digest = spin_common::sha256::hex_digest_from_bytes(&fake_file);
235        cache.write_data(&fake_file, &fake_file_digest).await?;
236
237        let mut component = LockedComponent {
238            id: component_id.into(),
239            metadata: Default::default(),
240            source: spin_locked_app::locked::LockedComponentSource {
241                content_type: "application/wasm".into(),
242                content: ContentRef {
243                    source: None,
244                    inline: Some(fake_wasm),
245                    digest: Some(fake_wasm_digest),
246                },
247            },
248            env: Default::default(),
249            files: vec![ContentPath {
250                content: ContentRef {
251                    source: None,
252                    inline: Some(fake_file),
253                    digest: Some(fake_file_digest),
254                },
255                path: "file.txt".into(),
256            }],
257            config: Default::default(),
258            dependencies: Default::default(),
259            trigger_dependencies: Default::default(),
260            host_requirements: Default::default(),
261        };
262
263        loader
264            .resolve_component_content_refs(&mut component, &cache)
265            .await
266    }
