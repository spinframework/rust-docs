1//! Prepared application directories.
2//!
3//! A prepared directory holds everything a Spin application from a registry
4//! needs to run, laid out ahead of time so that `spin up --from-prepared` can
5//! run it in place, without copying its files or accessing a registry:
6//!
7//! ```text
8//! <dir>/
9//!   spin.prepared.lock      the locked app, with sources relative to <dir>
10//!   wasm/<sha256 hex>.wasm  component, dependency and trigger dependency Wasm
11//!   assets/<component id>/  each component's files, as the guest sees them
12//! ```
13//!
14//! In the lock:
15//!
16//! - every Wasm source is exactly `wasm/<hex>.wasm`, where `<hex>` is the
17//!   SHA-256 of the file, and keeps its `sha256:<hex>` digest;
18//! - every file has an entry of its own, whose source is exactly
19//!   `assets/<component id>/<path>` and whose digest is the SHA-256 of the
20//!   file;
21//! - sources are relative paths with `/` separators, not URLs;
22//! - a component without files has no `assets/<component id>` directory;
23//! - the `origin` metadata names the artifact pinned to its digest, and
24//!   `prepared_layout_version` is the version of this layout.
25//!
26//! The directory contains no symbolic links.
27//!
28//! This is layout version 1. A Spin refuses a directory whose layout
29//! version it does not read, so a change to the layout comes with a new
30//! version.
31

62/// Writes the Spin application that a pull of `reference` resolved to
63/// `manifest` and `digest` (see [`Client::pull_with_digest`]) to `dir` as a
64/// prepared application.
65///
66/// `dir` must not exist or must be an empty directory, which is replaced (so
67/// it cannot be a mount point). The application is written to a temporary
68/// directory next to `dir` and moved into place at the end, so that a
69/// failure leaves nothing half-written.
70pub async fn prepare(
71    client: &Client,
72    reference: &str,
73    manifest: &OciImageManifest,
74    digest: &str,
75    dir: &Path,
76) -> Result<()> {
77    let lockfile_path = cached_lockfile_path(client, reference, digest).await?;
78    let origin = pinned_reference(reference, digest)?;
79    let Some(app) = read_locked_app(manifest, &lockfile_path, &origin).await? else {
80        bail!("{reference:?} is a Wasm package, not a Spin application, and cannot be prepared");
81    };
82    write_prepared(app, &client.cache, dir).await
83}

85/// Checks that `dir` names a directory that is missing or empty, and is not a
86/// symbolic link, so that an application can be prepared in it.
87pub fn check_target(dir: &Path) -> Result<()> {
88    ensure!(
89        dir.file_name().is_some(),
90        "{} does not name a directory to prepare the app in",
91        quoted_path(dir)
92    );
93    let metadata = match std::fs::symlink_metadata(dir) {
94        Ok(metadata) => metadata,
95        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(()),
96        Err(e) => return Err(e).with_context(|| format!("cannot use {}", quoted_path(dir))),
97    };
98    ensure!(
99        !metadata.file_type().is_symlink(),
100        "{} is a symbolic link",
101        quoted_path(dir)
102    );
103    ensure!(metadata.is_dir(), "{} is not a directory", quoted_path(dir));
104    let mut entries =
105        std::fs::read_dir(dir).with_context(|| format!("cannot read {}", quoted_path(dir)))?;
106    ensure!(
107        entries.next().is_none(),
108        "{} is not empty",
109        quoted_path(dir)
110    );
111    Ok(())
112}

114/// Writes a locked app read from the registry cache to `dir` as a prepared
115/// application.
116async fn write_prepared(mut app: LockedApp, cache: &Cache, dir: &Path) -> Result<()> {
117    check_target(dir)?;
118    validate_lock(&app)?;
119
120    let staging = create_staging_dir(dir)?;
121    for component in &mut app.components {
122        let id = component.id.clone();
123        prepare_component(component, cache, staging.path())
124            .await
125            .with_context(|| format!("failed to prepare component {id:?}"))?;
126    }
127    write_lock(&mut app, staging.path()).await?;
128
129    // On failure, dropping `staging` removes it.
130    move_into_place(staging.path(), dir).await?;
131    // `staging` is now `dir`, which must be kept.
132    let _ = staging.keep();
133    Ok(())
134}

136/// The artifact reference pinned to its digest, keeping the tag if any.
137fn pinned_reference(reference: &str, digest: &str) -> Result<String> {
138    let parsed = Reference::try_from(reference).context("invalid reference")?;
139    let (registry, repository) = (parsed.registry().to_owned(), parsed.repository().to_owned());
140    let pinned = match parsed.tag() {
141        Some(tag) => {
142            Reference::with_tag_and_digest(registry, repository, tag.to_owned(), digest.to_owned())
143        }
144        None => Reference::with_digest(registry, repository, digest.to_owned()),
145    };
146    Ok(pinned.whole())
147}

149/// Creates the directory to write the prepared app in. It is next to `dir`,
150/// so that it can be renamed to `dir`.
151fn create_staging_dir(dir: &Path) -> Result<tempfile::TempDir> {
152    let parent = match dir.parent() {
153        Some(parent) if !parent.as_os_str().is_empty() => parent,
154        _ => Path::new("."),
155    };
156    std::fs::create_dir_all(parent)
157        .with_context(|| format!("failed to create {}", quoted_path(parent)))?;
158    tempfile::Builder::new()
159        .prefix(".spin-prepare-")
160        .tempdir_in(parent)
161        .with_context(|| format!("failed to create a directory in {}", quoted_path(parent)))
162}

164/// Copies a component's Wasm and files to the prepared directory `dir`, and
165/// points their sources there.
166async fn prepare_component(
167    component: &mut LockedComponent,
168    cache: &Cache,
169    dir: &Path,
170) -> Result<()> {
171    copy_wasm(component, cache, dir).await?;
172    copy_files(component, cache, dir).await
173}
174
175/// Copies the component's Wasm, and that of its dependencies and trigger
176/// dependencies, from the cache to `wasm/<hex>.wasm`. Wasm used more than once
177/// is copied once.
178///
179/// The cache is not checked when it is read, so each copy is checked against
180/// its digest: the lock must record what the directory holds.
181async fn copy_wasm(component: &mut LockedComponent, cache: &Cache, dir: &Path) -> Result<()> {
182    for content in wasm_content_refs_mut(component) {
183        let digest = content_digest(content)?.to_owned();
184        let source = wasm_source(&digest)?;
185        let dest = dir.join(&source);
186        if !dest.exists() {
187            let cached = cache.wasm_file(&digest)?;
188            copy_file(&cached, &dest).await?;
189            ensure!(
190                matches_digest(&dest, &digest)?,
191                "{} in the cache does not match its digest {digest}",
192                quoted_path(&cached)
193            );
194        }
195        *content = prepared_content(source, digest);
196    }
197    Ok(())
198}
199
200/// Writes the component's files to `assets/<component id>/`, as the guest
201/// sees them, and records the source and digest of each.
202///
203/// As for Wasm, each file with a digest is checked against it once written.
204/// Inline files may have no digest, and then theirs is computed.
205async fn copy_files(component: &mut LockedComponent, cache: &Cache, dir: &Path) -> Result<()> {
206    write_component_files(
207        &component.id,
208        &component.files,
209        cache,
210        &dir.join(ASSETS_DIR),
211    )
212    .await?;
213    for file in &mut component.files {
214        let source = asset_source(&component.id, &file.path);
215        let written = dir.join(&source);
216        let digest = match &file.content.digest {
217            Some(digest) => {
218                ensure!(
219                    matches_digest(&written, digest)?,
220                    "file {} does not match its digest {digest}",
221                    quoted_path(&file.path)
222                );
223                digest.clone()
224            }
225            None => file_digest(&written)?,
226        };
227        file.content = prepared_content(source, digest);
228    }
229    Ok(())
230}

232/// The content reference to a file of the prepared directory.
233fn prepared_content(source: String, digest: String) -> ContentRef {
234    ContentRef {
235        source: Some(source),
236        inline: None,
237        digest: Some(digest),
238    }

241async fn copy_file(src: &Path, dest: &Path) -> Result<()> {
242    if let Some(parent) = dest.parent() {
243        tokio::fs::create_dir_all(parent)
244            .await
245            .with_context(|| format!("failed to create {}", quoted_path(parent)))?;
246    }
247    tokio::fs::copy(src, dest)
248        .await
249        .with_context(|| format!("failed to copy {}", quoted_path(src)))?;
250    Ok(())
251}

253/// Writes the lock, with the layout version, to the prepared directory `dir`.
254async fn write_lock(app: &mut LockedApp, dir: &Path) -> Result<()> {
255    app.metadata
256        .insert(LAYOUT_VERSION_KEY.to_owned(), LAYOUT_VERSION.into());
257    let path = dir.join(PREPARED_LOCK_FILE);
258    tokio::fs::write(&path, app.to_json()?)
259        .await
260        .with_context(|| format!("failed to write {}", quoted_path(&path)))
261}

263/// Moves the staged directory to `dir`, replacing it if it is an empty
264/// directory.
265async fn move_into_place(staged: &Path, dir: &Path) -> Result<()> {
266    // Elsewhere, rename replaces an empty directory atomically.
267    #[cfg(windows)]
268    if dir.exists() {
269        tokio::fs::remove_dir(dir)
270            .await
271            .with_context(|| format!("failed to replace {}", quoted_path(dir)))?;
272    }
273    tokio::fs::rename(staged, dir).await.with_context(|| {
274        format!(
275            "failed to move the prepared app to {} (it must be missing or an empty directory, and not a mount point)",
276            quoted_path(dir)
277        )
278    })
279}

281/// Checks the parts of a locked app that the prepared layout depends on:
282/// each component id names a directory, each file path names a file in it,
283/// and trigger dependencies belong to components.
284fn validate_lock(app: &LockedApp) -> Result<()> {
285    ensure!(
286        app.triggers
287            .iter()
288            .all(|trigger| trigger.trigger_dependencies.is_empty()),
289        "the app has trigger dependencies outside its components, which a prepared app cannot have"
290    );
291    check_ids_are_unique(app)?;
292    for component in &app.components {
293        let id = &component.id;
294        KebabId::try_from(id.clone())
295            .map_err(|reason| anyhow!("invalid component id {id:?}: {reason}"))?;
296        for file in &component.files {
297            check_file_path(&file.path).with_context(|| format!("component {id:?}"))?;
298        }
299        check_files_are_listed_once(component).with_context(|| format!("component {id:?}"))?;
300    }
301    Ok(())
302}
303
304fn check_ids_are_unique(app: &LockedApp) -> Result<()> {
305    let mut seen = HashSet::new();
306    for component in &app.components {
307        if !seen.insert(component.id.as_str()) {
308            bail!("component id {:?} is used more than once", component.id);
309        }
310    }
311    Ok(())
312}

314fn check_files_are_listed_once(component: &LockedComponent) -> Result<()> {
315    let mut seen = HashSet::new();
316    for file in &component.files {
317        if !seen.insert(file.path.as_path()) {
318            bail!("file {} is listed more than once", quoted_path(&file.path));
319        }
320    }
321    Ok(())
322}

324/// Checks that a file path names a file below the component's directory: a
325/// non-empty relative path of plain names.
326fn check_file_path(path: &Path) -> Result<()> {
327    // Locks record file paths with `/` separators. A backslash is a separator
328    // on Windows only, so it would name different files on different hosts.
329    ensure!(
330        path.components().next().is_some()
331            && is_safe_to_join(path)
332            && !path.to_string_lossy().contains('\\'),
333        "invalid file path {}",
334        quoted_path(path)
335    );
336    Ok(())
337}

339/// The source a prepared lock records for a Wasm file: `wasm/<hex>.wasm`,
340/// named after its SHA-256 digest.
341fn wasm_source(digest: &str) -> Result<String> {
342    let hex = digest
343        .strip_prefix("sha256:")
344        .filter(|hex| {
345            hex.len() == 64 && hex.bytes().all(|b| matches!(b, b'0'..=b'9' | b'a'..=b'f'))
346        })
347        .with_context(|| format!("unsupported Wasm digest {digest:?}: expected sha256:<hex>"))?;
348    Ok(format!("{WASM_DIR}/{hex}.wasm"))
349}

351/// The source a prepared lock records for a component file:
352/// `assets/<component id>/<path>`, with `/` separators. `path` must have
353/// passed [`check_file_path`].
354fn asset_source(component_id: &str, path: &Path) -> String {
355    let names: Vec<_> = path
356        .components()
357        .map(|name| name.as_os_str().to_string_lossy())
358        .collect();
359    format!("{ASSETS_DIR}/{component_id}/{}", names.join("/"))
360}

362/// The `sha256:<hex>` digest of a file.
363fn file_digest(path: &Path) -> Result<String> {
364    let hex = hex_digest_from_file(path)
365        .with_context(|| format!("failed to read {}", quoted_path(path)))?;
366    Ok(format!("sha256:{hex}"))
367}

369/// Whether the file matches its `sha256:<hex>` digest.
370fn matches_digest(path: &Path, digest: &str) -> Result<bool> {
371    Ok(file_digest(path)? == digest)
372}

374/// Reads the prepared application in `dir`, checks the directory against its
375/// lock, and returns a locked app that Spin can run from `dir` in place.
376///
377/// `dir` must contain no symbolic links. Each Wasm and file source in the
378/// lock must be the path the layout gives it, and each component's
379/// `assets/<component id>` directory, which is what the guest sees, must hold
380/// exactly its listed files. With `verify_digests`, every file is also
381/// checked against its digest. This reads the directory, and blocks while it
382/// does.
383///
384/// These checks hold when this function runs: Spin opens the files again
385/// later, so the caller must ensure that `dir` does not change while the app
386/// runs (for example, a read-only mount). The digests come from the lock in
387/// `dir`, so they detect a corrupted or partly changed directory, not a
388/// different application: trust a prepared directory as you would the code
389/// in it. Hard links are not detected.
390///
391/// In the returned app, Wasm sources are `file:` URLs into `dir`, and each
392/// component's files become one mount of `assets/<component id>` at `/`, as
393/// for apps loaded from a registry. Nothing is written to `dir`.
394pub fn load(dir: &Path, verify_digests: bool) -> Result<LockedApp> {
395    let mut prepared = PreparedDir::scan(dir, verify_digests)?;
396    let mut app = prepared.read_lock()?;
397    validate_lock(&app)?;
398    for component in &mut app.components {
399        let id = component.id.clone();
400        prepared
401            .load_component(component)
402            .with_context(|| format!("component {id:?} does not match the prepared directory"))?;
403    }
404    Ok(app)
405}

407/// A prepared directory being loaded.
408struct PreparedDir {
409    root: PathBuf,
410    /// The regular files in the directory, relative to it.
411    files: HashSet<PathBuf>,
412    /// The directories in the directory, relative to it.
413    dirs: HashSet<PathBuf>,
414    verify_digests: bool,
415    /// The Wasm files already checked against their digests.
416    checked_wasm: HashSet<String>,
417}
418
419impl PreparedDir {
420    /// Lists the prepared directory `dir`, which must contain only regular
421    /// files and directories. A symbolic link could point outside `dir`, and
422    /// the guest would see what it points to.
423    fn scan(dir: &Path, verify_digests: bool) -> Result<Self> {
424        let root = std::fs::canonicalize(dir)
425            .with_context(|| format!("cannot use {}", quoted_path(dir)))?;
426        let mut files = HashSet::new();
427        let mut dirs = HashSet::new();
428        for entry in walkdir::WalkDir::new(&root).min_depth(1) {
429            let entry = entry.with_context(|| format!("cannot read {}", quoted_path(&root)))?;
430            let path = entry.path().strip_prefix(&root)?.to_owned();
431            let file_type = entry.file_type();
432            if file_type.is_file() {
433                files.insert(path);
434            } else if file_type.is_dir() {
435                dirs.insert(path);
436            } else if file_type.is_symlink() {
437                bail!(
438                    "{} is a symbolic link, which a prepared directory cannot contain",
439                    quoted_path(entry.path())
440                );
441            } else {
442                bail!(
443                    "{} is not a regular file or a directory",
444                    quoted_path(entry.path())
445                );
446            }
447        }
448        Ok(Self {
449            root,
450            files,
451            dirs,
452            verify_digests,
453            checked_wasm: HashSet::new(),
454        })
455    }
456
457    /// Reads the lock and checks its layout version.
458    fn read_lock(&self) -> Result<LockedApp> {
459        ensure!(
460            self.has_file(PREPARED_LOCK_FILE),
461            "{} is not a prepared application: it has no {PREPARED_LOCK_FILE}",
462            quoted_path(&self.root)
463        );
464        let path = self.root.join(PREPARED_LOCK_FILE);
465        let lock = std::fs::read(&path)
466            .with_context(|| format!("failed to read {}", quoted_path(&path)))?;
467        let app = LockedApp::from_json(&lock)
468            .with_context(|| format!("failed to decode {}", quoted_path(&path)))?;
469        let version = app.metadata.get(LAYOUT_VERSION_KEY);
470        ensure!(
471            version.and_then(|v| v.as_u64()) == Some(LAYOUT_VERSION),
472            "{} has prepared layout version {}, but this Spin reads version {LAYOUT_VERSION}",
473            quoted_path(&path),
474            version.map_or("(none)".to_owned(), |v| v.to_string())
475        );
476        Ok(app)
477    }
478
479    /// Checks a component against the directory, and points its sources
480    /// into it.
481    fn load_component(&mut self, component: &mut LockedComponent) -> Result<()> {
482        self.load_wasm(component)?;
483        self.load_files(component)
484    }
485
486    /// Checks that the component's Wasm files are where the layout puts
487    /// them, and points their sources at them. Wasm used more than once is
488    /// checked against its digest once.
489    fn load_wasm(&mut self, component: &mut LockedComponent) -> Result<()> {
490        for content in wasm_content_refs_mut(component) {
491            let digest = content_digest(content)?.to_owned();
492            let source = wasm_source(&digest)?;
493            check_source(content, &source)?;
494            ensure!(
495                self.has_file(&source),
496                "{} is missing",
497                quoted_path(&source)
498            );
499            let path = self.root.join(&source);
500            if self.verify_digests && self.checked_wasm.insert(source) {
501                check_digest(&path, &digest)?;
502            }
503            *content = ContentRef {
504                source: Some(file_url(&path)?),
505                ..Default::default()
506            };
507        }
508        Ok(())
509    }
510
511    /// Checks that the component's directory holds exactly its listed files,
512    /// and replaces them with one mount of that directory, as the guest sees
513    /// it.
514    fn load_files(&self, component: &mut LockedComponent) -> Result<()> {
515        let mut sources = Vec::with_capacity(component.files.len());
516        for file in &component.files {
517            let source = asset_source(&component.id, &file.path);
518            check_source(&file.content, &source)?;
519            let digest = content_digest(&file.content)?.to_owned();
520            sources.push((source, digest));
521        }
522
523        let component_dir = Path::new(ASSETS_DIR).join(&component.id);
524        let expected = sources
525            .iter()
526            .map(|(source, _)| Path::new(source))
527            .collect();
528        self.check_holds_exactly(&component_dir, &expected)?;
529        if self.verify_digests {
530            for (source, digest) in &sources {
531                check_digest(&self.root.join(source), digest)?;
532            }
533        }
534
535        if !component.files.is_empty() {
536            component.files = vec![ContentPath {
537                content: ContentRef {
538                    source: Some(file_url(&self.root.join(&component_dir))?),
539                    ..Default::default()
540                },
541                path: "/".into(),
542            }];
543        }
544        Ok(())
545    }
546
547    fn has_file(&self, path: impl AsRef<Path>) -> bool {
548        self.files.contains(path.as_ref())
549    }
550
551    /// Checks that `component_dir` holds exactly the `expected` files and
552    /// the directories that lead to them. With no files, `component_dir`
553    /// must not exist.
554    fn check_holds_exactly(&self, component_dir: &Path, expected: &HashSet<&Path>) -> Result<()> {
555        if expected.is_empty() {
556            ensure!(
557                !self.files.contains(component_dir) && !self.dirs.contains(component_dir),
558                "{} was not expected, because the component has no files",
559                quoted_path(component_dir)
560            );
561            return Ok(());
562        }
563        for file in expected {
564            ensure!(self.has_file(file), "{} is missing", quoted_path(file));
565        }
566
567        let expected_dirs: HashSet<&Path> = expected
568            .iter()
569            .flat_map(|file| {
570                file.ancestors()
571                    .skip(1)
572                    .take_while(|ancestor| ancestor.starts_with(component_dir))
573            })
574            .collect();
575        let unexpected_files = self
576            .files
577            .iter()
578            .filter(|file| file.starts_with(component_dir) && !expected.contains(file.as_path()));
579        let unexpected_dirs = self
580            .dirs
581            .iter()
582            .filter(|dir| dir.starts_with(component_dir) && !expected_dirs.contains(dir.as_path()));
583        if let Some(path) = unexpected_files.chain(unexpected_dirs).next() {
584            bail!(
585                "{} is not listed in the lock, and the guest would see it",
586                quoted_path(path)
587            );
588        }
589        Ok(())
590    }
591}

593/// Checks that the lock records `expected` as the source of `content`.
594fn check_source(content: &ContentRef, expected: &str) -> Result<()> {
595    ensure!(
596        content.source.as_deref() == Some(expected),
597        "source {:?} should be {expected:?}",
598        content.source.as_deref().unwrap_or_default()
599    );
600    Ok(())
601}

603/// Checks that the file matches its digest.
604fn check_digest(path: &Path, digest: &str) -> Result<()> {
605    ensure!(
606        matches_digest(path, digest)?,
607        "{} does not match its digest {digest}",
608        quoted_path(path)
609    );
610    Ok(())
611}

613fn file_url(path: &Path) -> Result<String> {
614    reqwest::Url::from_file_path(path)
615        .map(|url| url.to_string())
616        .map_err(|_| anyhow!("cannot make a file URL of {}", quoted_path(path)))
617}

716    fn all_sources(app: &mut LockedApp) -> Vec<String> {

1029    /// Makes a directory tree read-only, and writable again when dropped.
1030    #[cfg(unix)]
1031    struct ReadOnly<'a>(&'a Path);
1032
1033    #[cfg(unix)]
1034    impl<'a> ReadOnly<'a> {
1035        fn new(dir: &'a Path) -> Result<Self> {
1036            Self::set_modes(dir, 0o444, 0o555)?;
1037            Ok(Self(dir))
1038        }
1039
1040        fn set_modes(dir: &Path, file_mode: u32, dir_mode: u32) -> Result<()> {
1041            use std::os::unix::fs::PermissionsExt;
1042            // Directories last when locking, first when unlocking.
1043            let lock = dir_mode & 0o200 == 0;
1044            for entry in walkdir::WalkDir::new(dir).contents_first(lock) {
1045                let entry = entry?;
1046                let mode = if entry.file_type().is_dir() {
1047                    dir_mode
1048                } else {
1049                    file_mode
1050                };
1051                std::fs::set_permissions(entry.path(), std::fs::Permissions::from_mode(mode))?;
1052            }
1053            Ok(())
1054        }
1055    }
1056
1057    #[cfg(unix)]
1058    impl Drop for ReadOnly<'_> {
1059        fn drop(&mut self) {
1060            let _ = Self::set_modes(self.0, 0o644, 0o755);
1061        }
1062    }
